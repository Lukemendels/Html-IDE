    function syncLibraryIndicators() {
      const code = state.code;
      const libIds = [
        'lib-alpine', 'lib-picocss', 'lib-jszip', 'lib-docxtemplater',
        'lib-docx', 'lib-mammoth', 'lib-sheetjs', 'lib-pptxgen', 'lib-pdfjs'
      ];

      let activeCount = 0;
      libIds.forEach(libId => {
        const hasStem = code.includes(`id="${libId}-stem"`);
        const dot = document.getElementById(`dot-${libId.replace('lib-', '')}`);
        const row = dot ? dot.closest('.lib-row') : null;

        if (dot) {
          if (hasStem) {
            dot.className = "lib-status-dot h-1.5 w-1.5 rounded-full bg-emerald-500 animate-pulse shrink-0 shadow-lg shadow-emerald-500/50";
            if (row) {
              row.classList.add('border-emerald-500/30', 'bg-emerald-500/5');
              row.classList.remove('border-[#222227]', 'bg-[#141417]/50');
            }
            activeCount++;
          } else {
            dot.className = "lib-status-dot h-1.5 w-1.5 rounded-full bg-gray-600 shrink-0";
            if (row) {
              row.classList.remove('border-emerald-500/30', 'bg-emerald-500/5');
              row.classList.add('border-[#222227]', 'bg-[#141417]/50');
            }
          }
        }
      });

      if (activeCount > 0) {
        elements.libPackerCount.textContent = `${activeCount} active`;
        elements.libPackerCount.classList.remove('hidden');
      } else {
        elements.libPackerCount.classList.add('hidden');
      }
    }
